cmake_minimum_required(VERSION 3.22.1)
project(llama_android C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../llama-revision.txt" LLAMA_PINNED_COMMIT LIMIT_COUNT 1)
if(NOT LLAMA_PINNED_COMMIT MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Invalid llama-revision.txt")
endif()

execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." OUTPUT_VARIABLE JARVIS_GIT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE JARVIS_GIT_RESULT)
if(NOT JARVIS_GIT_RESULT EQUAL 0 OR JARVIS_GIT_COMMIT STREQUAL "")
    set(JARVIS_GIT_COMMIT "unknown")
endif()

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt")
    message(STATUS "llama.cpp not found. Cloning pinned commit ${LLAMA_PINNED_COMMIT}...")
    execute_process(COMMAND git clone https://github.com/ggml-org/llama.cpp.git ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp RESULT_VARIABLE GIT_CLONE_RESULT)
    if(NOT GIT_CLONE_RESULT EQUAL 0)
        message(FATAL_ERROR "Failed to clone official llama.cpp repository")
    endif()
    execute_process(COMMAND git checkout ${LLAMA_PINNED_COMMIT} WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp RESULT_VARIABLE GIT_CHECKOUT_RESULT)
    if(NOT GIT_CHECKOUT_RESULT EQUAL 0)
        message(FATAL_ERROR "Failed to checkout pinned llama.cpp commit ${LLAMA_PINNED_COMMIT}")
    endif()
endif()

execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp" OUTPUT_VARIABLE LLAMA_ACTUAL_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE LLAMA_REV_RESULT)
if(NOT LLAMA_REV_RESULT EQUAL 0 OR NOT LLAMA_ACTUAL_COMMIT STREQUAL LLAMA_PINNED_COMMIT)
    message(FATAL_ERROR "llama.cpp cache revision ${LLAMA_ACTUAL_COMMIT} differs from ${LLAMA_PINNED_COMMIT}. Refresh the dependency checkout.")
endif()
execute_process(COMMAND git status --porcelain --untracked-files=no WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp" OUTPUT_VARIABLE LLAMA_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT LLAMA_DIRTY STREQUAL "")
    message(FATAL_ERROR "llama.cpp has modified tracked files; refusing misleading build provenance")
endif()

set(GGML_ALL_QUANTS ON CACHE BOOL "Enable all quantization types including the actual mobile-QAT qtypes" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Disable host-native tuning for Android cross compilation" FORCE)
set(GGML_NEON ON CACHE BOOL "Enable ARM NEON" FORCE)
set(GGML_ARM_F16C ON CACHE BOOL "Enable ARM F16C" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "Disable OpenMP" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "Disable llamafile on Android" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "Disable examples" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "Disable llama.cpp tests" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "Disable server" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "Build common chat template support" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "Build shared llama/ggml libraries" FORCE)

add_subdirectory(llama.cpp)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(JARVIS_CPU_AUDIT_PATCH "${CMAKE_CURRENT_SOURCE_DIR}/../../../../scripts/patch-llama-cpu-jarvis-audit.py")
set(JARVIS_CPU_AUDIT_RUNNER "${CMAKE_CURRENT_SOURCE_DIR}/../../../../scripts/run-llama-cpu-jarvis-audit.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${JARVIS_CPU_AUDIT_PATCH}" "${JARVIS_CPU_AUDIT_RUNNER}")
file(SHA256 "${JARVIS_CPU_AUDIT_PATCH}" JARVIS_CPU_AUDIT_PATCH_SHA256)
set(JARVIS_CPU_AUDIT_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/jarvis-cpu-audit-src")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${JARVIS_CPU_AUDIT_RUNNER}" "${JARVIS_CPU_AUDIT_PATCH}" "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp" "${JARVIS_CPU_AUDIT_SOURCE_DIR}" RESULT_VARIABLE JARVIS_CPU_PATCH_RESULT OUTPUT_VARIABLE JARVIS_CPU_PATCH_STDOUT ERROR_VARIABLE JARVIS_CPU_PATCH_STDERR)
if(NOT JARVIS_CPU_PATCH_RESULT EQUAL 0)
    message(FATAL_ERROR "Jarvis CPU audit source patch failed: ${JARVIS_CPU_PATCH_STDERR}")
endif()
if(NOT EXISTS "${JARVIS_CPU_AUDIT_SOURCE_DIR}/ggml-cpu.c" OR NOT EXISTS "${JARVIS_CPU_AUDIT_SOURCE_DIR}/ggml-cpu.cpp")
    message(FATAL_ERROR "Jarvis CPU audit patch did not produce both CPU translation units")
endif()

set(JARVIS_CANONICAL_CPU_C "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/ggml/src/ggml-cpu/ggml-cpu.c")
set(JARVIS_CANONICAL_CPU_CPP "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/ggml/src/ggml-cpu/ggml-cpu.cpp")
set_source_files_properties("${JARVIS_CANONICAL_CPU_C}" "${JARVIS_CANONICAL_CPU_CPP}" TARGET_DIRECTORY ggml-cpu PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(ggml-cpu PRIVATE "${JARVIS_CPU_AUDIT_SOURCE_DIR}/ggml-cpu.c" "${JARVIS_CPU_AUDIT_SOURCE_DIR}/ggml-cpu.cpp")
target_include_directories(ggml-cpu PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/ggml/src/ggml-cpu")
# The generated C audit uses C11 atomic_* types/functions. Force-include the standard atomic
# declarations for C only so the diagnostic copy stays valid even though the pinned source
# itself does not include <stdatomic.h>.
target_compile_options(ggml-cpu PRIVATE "$<$<COMPILE_LANGUAGE:C>:-include>" "$<$<COMPILE_LANGUAGE:C>:stdatomic.h>")
target_compile_definitions(ggml-cpu PRIVATE JARVIS_LLAMA_COMMIT="${LLAMA_ACTUAL_COMMIT}" JARVIS_CPU_AUDIT_PATCH_SHA256="${JARVIS_CPU_AUDIT_PATCH_SHA256}")

add_library(llama_android SHARED llama-android-accelerated.cpp htp-execution-audit.cpp htp-audit-interpose.cpp native-crash-handler.cpp accelerator-bootstrap.cpp common-json-compat.cpp)
target_include_directories(llama_android PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/include ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/ggml/include ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/src ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/common)
find_library(log-lib log)
find_library(android-lib android)
find_library(dl-lib dl)
target_link_libraries(llama_android PRIVATE llama ggml llama-common ${log-lib} ${android-lib} ${dl-lib})
target_link_options(llama_android PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384" "-Wl,--wrap=llama_log_set" "-Wl,--wrap=llama_init_from_model" "-Wl,--wrap=llama_decode" "-Wl,--wrap=llama_sampler_sample")
target_compile_definitions(llama_android PRIVATE JARVIS_LLAMA_COMMIT="${LLAMA_ACTUAL_COMMIT}" JARVIS_GIT_COMMIT="${JARVIS_GIT_COMMIT}")
